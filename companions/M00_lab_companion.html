checks = {}

def record(item, status, detail, fix=""):
    """Take one check result (item, PASS/FIX/NOTE, detail, fix advice); print it and keep it."""
    # A passed check needs no advice.
    if status == "PASS":
        fix = ""

    # ljust(4) pads the status to four characters, so the | lines up.

    if fix:
        print("       What to do:", fix)

1.3 Required packages

MINIMUM = {"numpy": "1.24", "pandas": "3.0", "scikit-learn": "1.8", "scipy": "1.10", "matplotlib": "3.7"}
FIX_PACKAGES = ("Check that the kernel is ban6403 (step 1.2). Then, in Anaconda Prompt (Windows) or Terminal (macOS), "
                "go to the repository folder and run: conda env update -n ban6403 -f environment.yml --prune")

def major_minor(number):
    """Take a version text such as '3.0.5' and return (3, 0), so two versions compare as numbers."""
    # re.findall finds every run of digits: '3.0.5' gives ['3', '0', '5'].
    parts = re.findall(r"\d+", number)
    return (int(parts[0]), int(parts[1]))

# Keep the versions for the summary file in Part 3.
versions = {"python": platform.python_version()}
for package, minimum in MINIMUM.items():
    # version() raises PackageNotFoundError when a package is missing; that is a FIX, not a crash.
    try:
        installed = version(package)
    except PackageNotFoundError:
        record("package " + package, "FIX", "not installed", FIX_PACKAGES)
        continue
    versions[package] = installed
    ok = major_minor(installed) >= major_minor(minimum)
    record("package " + package, "PASS" if ok else "FIX",
           f"version {installed} (need {minimum} or newer)", FIX_PACKAGES)

1.5 Data files: fingerprints

A SHA-256 hash is a 64-character fingerprint computed from a file's bytes. If one character changes, the hash changes. text_sha256 treats Windows and Mac line endings the same way.

EXPECTED_SHA256 = {
    "data/raw/customer_feedback_m01.csv": "de9a95b7aec38238388de50139cfa2419d0b5549ce9ba39b90bdee060156ab40",
    "data/partitions/m02_partition.csv": "2a08b6a84ba16ddc1c2225ab64936f03164c4192fa959aa7ab05b861c389189c",
    "data/course/cfpb_complaints.csv": "f6a7bfdcae91598b1e218bdb08d018da357b32715e45ba90d87bc9ff738080d7",
    "data/course/clothing_reviews.csv": "8a61265cd4b1421c95949affbc20acb39bbbbb44748daca78b11a076262e8682",
}
FIX_DATA = ("Do not edit files in data/. In GitHub Desktop, open the Changes tab. If the file is listed, right-click it "
            "and choose Discard Changes. If it is not listed, download the file again from the course template on GitHub "
            "and put it back in the same folder.")

def text_sha256(path):
    """Take a file path and return its SHA-256 fingerprint, counting Windows line endings as Mac ones."""
    # read_bytes() reads the raw file; b"\r\n" is a Windows line ending.
    data = Path(path).read_bytes().replace(b"\r\n", b"\n")
    return hashlib.sha256(data).hexdigest()

for name, expected in EXPECTED_SHA256.items():
    path = REPO / name
    if path.exists():
        same = text_sha256(path) == expected
        # expected[:12] shows the first 12 characters of the fingerprint.
        if same:
            detail = "sha256 " + expected[:12] + " matches"
        else:
            detail = "sha256 " + expected[:12] + " does NOT match"
    else:
        same, detail = False, "not found"
    record("file " + path.name, "PASS" if same else "FIX", detail, FIX_DATA)

1.7 Cached embeddings

Embeddings are lists of numbers that describe the meaning of a text. The labs load saved embeddings from data/embeddings/, so you do not download or run a language model. Each file has a manifest, a small JSON file with its size and fingerprint. The cell checks a fixed list of the eight files the labs need.

# The eight saved embedding files that the labs use.
EMBEDDINGS = ["cfpb_chunks_minilm", "cfpb_minilm", "clothing_minilm", "feedback_minilm",
              "m05_sentences_minilm", "m06_queries_minilm", "m12_passages_minilm", "m12_questions_minilm"]
EMBEDDINGS_DIR = REPO / "data/embeddings"

for name in EMBEDDINGS:
    manifest = EMBEDDINGS_DIR / (name + "_manifest.json")
    if not ready:
        record("embeddings " + name + ".npy", "SKIP", SKIPPED)
        continue
    if not manifest.exists():
        record("embeddings " + name + ".npy", "FIX", "manifest not found", FIX_DATA)
        continue

    # Read the manifest into a dictionary: file name, rows, dimensions, fingerprint.
    info = json.loads(manifest.read_text())
    vectors_file = EMBEDDINGS_DIR / info["cache"]

    # Does the saved file exist and match its fingerprint?
    same = False
    if vectors_file.exists():
        same = hashlib.sha256(vectors_file.read_bytes()).hexdigest() == info["cache_sha256"]

    # np.load reads the array; its shape must be (rows, numbers per vector).
    ok = same and np.load(vectors_file).shape == (info["rows"], info["dimensions"])

    if same:
        hash_text = "matches"
    else:
        hash_text = "does NOT match"
    detail = f"{info['rows']:,} vectors of {info['dimensions']} numbers, sha256 {hash_text}"
    record("embeddings " + info["cache"], "PASS" if ok else "FIX", detail, FIX_DATA)

2.3 Missing text and text length

describe() summarizes a number column. Its rows are count (how many values), mean (average) and std (standard deviation, the typical distance from the mean). Then come min, the quartiles 25%, 50% and 75%, and max. The 50% row is the median.

print("messages with no text:", feedback["text"].isna().sum())
# fillna("") replaces the missing text with an empty string, so split() works on every row.
# .str applies a text method to every row: split() makes a list of words, len() counts them.
feedback["n_words"] = feedback["text"].fillna("").str.split().str.len()
feedback["n_words"].describe().round(1)

Codespaces

Common Problems

Read the last line of the error message first. Then find it below.

Companion-added instructor note

FIX | notebook kernel, or ModuleNotFoundError: No module named 'sklearn'

The notebook runs in the wrong Python. This is the most common setup problem. To recover:

  1. Click the kernel name at the top right of the notebook.
  2. Choose Select Another Kernel > Python Environments > ban6403.
  3. Choose Restart, then Run All.
  4. Step 1.2 should now show PASS | notebook kernel: environment ban6403.

If ban6403 is not in the list, the environment was not created. Run conda env create -f environment.yml in the repository folder, then run Developer: Reload Window from the VS Code command palette (Ctrl+Shift+P or Cmd+Shift+P).

Companion-added instructor note

'conda' is not recognized (Windows) or conda: command not found (macOS)

On Windows, use Anaconda Prompt from the Start menu, not the ordinary Command Prompt. On macOS, close Terminal and open it again after installing Anaconda. If it still fails, run the Anaconda installer again.

Companion-added instructor note

A package shows FIX

First check the kernel (step 1.2). If the kernel is ban6403, update the environment: in Anaconda Prompt or Terminal, go to the repository folder and run conda env update -n ban6403 -f environment.yml --prune. Then restart the kernel and run from the top.

Companion-added instructor note

A folder shows FIX: not found

VS Code has only the notebook open, not the repository folder. Choose File > Open Folder, select the repository folder, and open the notebook from notebooks/.

Companion-added instructor note

FIX | Git repository: this folder has no Git history

The folder was probably downloaded as a ZIP file. Clone your copy with GitHub Desktop (File > Clone Repository), open the cloned folder in VS Code, and run the notebook there.

Companion-added instructor note

A data file or embedding shows does NOT match

The file differs from the course version. Do not edit files in data/. In GitHub Desktop, open the Changes tab. If the file is listed, right-click it and choose Discard Changes. If it is not listed, download the file again from the course template on GitHub and put it back in the same folder. Opening a CSV file in Excel and saving it is a common cause.

Companion-added instructor note

NOTE | folder location: inside OneDrive

Cloud sync can lock or duplicate files while Git works. If everything else passes, you can continue. If you see odd errors later, clone the repository again into a normal folder such as Documents/GitHub.

Companion-added instructor note

Run All stops at step 1.6 with ModuleNotFoundError: No module named 'pandas'

Step 1.3 already showed a FIX for pandas. Fix the kernel or the environment as described above, then run from the top.

Companion-added instructor note

SKIP lines in steps 1.6 and 1.7, then NameError: name 'feedback' is not defined in step 2.1

A package or data file check in step 1.3 or 1.5 needs a fix, so the data was not read. Fix the FIX rows shown in step 1.8, then choose Restart and Run All.
